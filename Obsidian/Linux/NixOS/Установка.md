```
mkdir -p ~/nixos-config/hosts/nixos && cd ~/nixos-config && \
cat > flake.nix <<'EOF'
{
  description = "NixOS configuration";

  inputs = {
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-26.05";

    disko = {
      url = "github:nix-community/disko";
      inputs.nixpkgs.follows = "nixpkgs";
    };
  };

  outputs = { self, nixpkgs, disko, ... }:
    let
      system = "x86_64-linux";
      pkgs = nixpkgs.legacyPackages.${system};
    in
    {
      nixosConfigurations.nixos = nixpkgs.lib.nixosSystem {
        inherit system;

        modules = [
          ./hosts/nixos/configuration.nix
          ./disko.nix
          disko.nixosModules.disko
        ];
      };

      apps.${system}.disko = {
        type = "app";
        program = "${disko.packages.${system}.disko}/bin/disko";
      };
    };
}
EOF

cat > disko.nix <<'EOF'
{
  disko.devices = {
    disk.main = {
      type = "disk";
      device = "/dev/vda";

      content = {
        type = "gpt";

        partitions = {
          ESP = {
            size = "1G";
            type = "EF00";

            content = {
              type = "filesystem";
              format = "vfat";
              mountpoint = "/boot";
              mountOptions = [ "umask=0077" ];
            };
          };

          root = {
            size = "100%";

            content = {
              type = "filesystem";
              format = "ext4";
              mountpoint = "/";
            };
          };
        };
      };
    };
  };
}
EOF

cat > hosts/nixos/configuration.nix <<'EOF'
{ config, pkgs, ... }:

{
  system.stateVersion = "26.05";
  
  #Добавление драйверов virtio
  boot.initrd.availableKernelModules = [
    "virtio_pci"
    "virtio_blk"
  ];
  
  boot.loader.systemd-boot.enable = true;
  boot.loader.efi.canTouchEfiVariables = true;

  networking.hostName = "nixos";
  networking.networkmanager.enable = true;

  time.timeZone = "Europe/Moscow";

  i18n.defaultLocale = "ru_RU.UTF-8";
  console.keyMap = "ru";

  users.users.nikolaev = {
    isNormalUser = true;
    extraGroups = [ "wheel" ];
    initialPassword = "klibrudak";
  };

  security.sudo.wheelNeedsPassword = true;

  environment.systemPackages = with pkgs; [
    vim
    git
    curl
    wget
    htop
  ];

  services.openssh.enable = true;

  networking.firewall.allowedTCPPorts = [
    22
  ];
}
EOF

sudo nix --extra-experimental-features 'nix-command flakes' flake lock && \
sudo nix --extra-experimental-features 'nix-command flakes' run .#disko -- --mode disko ./disko.nix
sudo nixos-install --flake .#nixos
```
