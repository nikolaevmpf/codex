
mkdir -p ~/nixos-config/hosts/nixos
cd ~/nixos-config

```
cat > ~/nixos-config/flake.nix <<'EOF'
{
  description = "NixOS home desktop";

  inputs = {
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-26.05";

    disko = {
      url = "github:nix-community/disko";
      inputs.nixpkgs.follows = "nixpkgs";
    };
  };

  outputs = { self, nixpkgs, disko, ... }:
    {
      nixosConfigurations.nixos = nixpkgs.lib.nixosSystem {
        system = "x86_64-linux";

        modules = [
          ./hosts/nixos/configuration.nix
          disko.nixosModules.disko
        ];
      };
    };
}
EOF
```


```
cat > ~/nixos-config/hosts/nixos/configuration.nix <<'EOF'
{ config, pkgs, ... }:

  fileSystems."/" = {
    device = "/dev/disk/by-uuid/42e98a3a-e4c8-4224-8b6f-f270e1f4b4e8";
    fsType = "ext4";
  };

  fileSystems."/boot" = {
    device = "/dev/disk/by-uuid/BBDC-A806";
    fsType = "vfat";
  };
  
  system.stateVersion = "26.05";

  # Flakes и современный интерфейс Nix
  nix.settings.experimental-features = [
    "nix-command"
    "flakes"
  ];

  # Загрузчик
  boot.loader.systemd-boot.enable = true;
  boot.loader.efi.canTouchEfiVariables = true;

  # Сеть
  networking.hostName = "nixos";
  networking.networkmanager.enable = true;

  # Время
  time.timeZone = "Europe/Moscow";

  # Локаль
  i18n.defaultLocale = "ru_RU.UTF-8";
  console.keyMap = "ru";

  # Пользователь
  users.users.nikolaev = {
    isNormalUser = true;

    extraGroups = [
      "wheel"
      "networkmanager"
    ];

    initialPassword = "klibrudak";
  };

  security.sudo.wheelNeedsPassword = true;

  # Базовые программы
  environment.systemPackages = with pkgs; [
    vim
    git
    curl
    wget
    htop
  ];

  # SSH
  services.openssh.enable = true;

  networking.firewall.allowedTCPPorts = [
    22
  ];
}
EOF
```

sudo nixos-rebuild build --flake .#nixos --option experimental-features 'nix-command flakes'

sudo nixos-rebuild switch --flake .#nixos --option experimental-features 'nix-command flakes'
